Audit membaca keluaran model apa adanya. Citation di luar retrieved evidence dicatat sebagai temuan evaluasi; tidak diganti otomatis. Untuk reproduksi, hentikan setiap langkah jika exit code bukan 0. Jangan menjalankan lock jika audit eksekusi gagal.

```powershell
$p8Run = "runtime/working/P8_final_v6_20260925"
$p4Evidence = "demo_evidence/ACQ-SIM-001_20260925_010848/artifacts/artifacts.csv"
python tools/audit_p8_run.py --experiment "$p8Run/experiment_output.json" --output "$p8Run/execution_audit.json"

python tools/lock_p8_outputs.py `
  --artifacts $p4Evidence `
  --baseline runtime/working/P5/baseline_findings.json `
  --config "$p8Run/run_config.json" `
  --experiment "$p8Run/experiment_output.json" `
  --run-log "$p8Run/run_log.jsonl" `
  --extra-file "$p8Run/input_code_manifest.json" `
  --extra-file "$p8Run/ollama_runner.py" `
  --extra-file "$p8Run/run_experiment.py" `
  --extra-file "$p8Run/retriever.py" `
  --extra-file runtime/working/P6/index.json `
  --extra-file runtime/working/P6/chunks.jsonl `
  --extra-file runtime/working/P5/p5_lock_manifest_v2.json `
  --extra-file "$p8Run/execution_audit.json" `
  --output "$p8Run/p8_lock_manifest.json"

python -m src.evaluation.evaluate_experiment `
  --artifacts $p4Evidence `
  --experiment "$p8Run/experiment_output.json" `
  --baseline runtime/working/P5/baseline_findings.json `
  --outputs-locked --lock-manifest "$p8Run/p8_lock_manifest.json" `
  --output "$p8Run/p9_precheck.json"

python -m src.report.build_report `
  --acquisition-manifest demo_evidence/ACQ-SIM-001_20260925_010848/acquisition_manifest.json `
  --artifact-manifest demo_evidence/ACQ-SIM-001_20260925_010848/artifacts/artifact_manifest.json `
  --evaluation "$p8Run/p9_precheck.json" `
  --p8-lock "$p8Run/p8_lock_manifest.json" `
  --config "$p8Run/run_config.json" `
  --output "$p8Run/run_report.md"
```
